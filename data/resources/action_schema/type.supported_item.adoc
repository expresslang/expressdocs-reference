TYPE supported_item = SELECT (action_directive, action, action_method);
END_TYPE;

(*"supported_item"
	The supported_item allows for the designation of an
	express_ref:[action_schema.action_directive],
	an express_ref:[action_schema.action],
	or an express_ref:[action_schema.action_method].

  NOTE: This specifies the use of an
	express_ref:[action_schema.action_resource].
*)
